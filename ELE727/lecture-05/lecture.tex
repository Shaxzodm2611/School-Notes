Recall: $Z_C=\frac{1}{sC}$ where $s=j\omega$

\begin{notederivation}{Transfer Function of a High-pass Filter}
    \frac{V_{\mathrm{out}}(s)}{V_{\mathrm{in}}(s)}
        &= \frac{R}{R+\frac{1}{sC}} \\
        &= \frac{sCR}{1+sCR}
\end{notederivation}
\keyequation{Transfer Function of a High-pass Filter}{
    |H(j\omega)| = \left|\frac{j\omega CR}{1+j\omega CR}\right|
}
